export async function GET(req:Request){const {searchParams}=new URL(req.url);const forceUserId=searchParams.get('userId');const u=await currentUser();if(!u)return NextResponse.json({error:"Giriş gerekli"},{status:401});const messages=await prisma.message.findMany({where:{OR:[{senderId:u.id},{recipientId:u.id}]},include:{sender:{select:{id:true,name:true,role:true}},recipient:{select:{id:true,name:true,role:true}}},orderBy:{createdAt:"desc"},take:1000});const cMap=new Map();messages.forEach(m=>{const other=m.senderId===u.id?m.recipient:m.sender;if(!cMap.has(other.id)){cMap.set(other.id,{id:other.id,name:other.name,subtitle:other.role==='PATIENT'?'Hasta':other.role==='DOCTOR'?'Uzman':other.role==='ORGANIZATION'?'Kurum':other.role==='AGENCY'?'Acente':'Kullanıcı',lastMessageAt:m.createdAt})}});if(forceUserId&&!cMap.has(forceUserId)&&forceUserId!==u.id){const target=await prisma.user.findUnique({where:{id:forceUserId},select:{id:true,name:true,role:true}});if(target){cMap.set(target.id,{id:target.id,name:target.name,subtitle:target.role==='PATIENT'?'Hasta':target.role==='DOCTOR'?'Uzman':target.role==='ORGANIZATION'?'Kurum':target.role==='AGENCY'?'Acente':'Kullanıcı',lastMessageAt:new Date()})}}const contacts=Array.from(cMap.values()).sort((a,b)=>new Date(b.lastMessageAt).getTime()-new Date(a.lastMessageAt).getTime());return NextResponse.json({contacts,messages:messages.map(m=>({id:m.id,body:m.body,createdAt:m.createdAt,readAt:m.readAt,senderId:m.senderId,senderName:m.sender.name,recipientId:m.recipientId,recipientName:m.recipient.name,mine:m.senderId===u.id})).reverse()})}
